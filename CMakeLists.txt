cmake_minimum_required(VERSION 3.25)
project(romeo LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -fno-exceptions)

add_executable(romeo-render renderer/main.cpp renderer/raster.cpp renderer/texture.cpp renderer/output.cpp)
target_include_directories(romeo-render PRIVATE core)
target_include_directories(romeo-render SYSTEM PRIVATE third_party/stb)
target_compile_options(romeo-render PRIVATE -O3)
find_package(Threads REQUIRED)
target_link_libraries(romeo-render PRIVATE Threads::Threads)

find_package(CURL REQUIRED)
add_library(yyjson STATIC third_party/yyjson/yyjson.c)
target_include_directories(yyjson PUBLIC third_party/yyjson)
target_compile_options(yyjson PRIVATE -O2 -w)
add_executable(romeo backend/main.cpp backend/wal.cpp backend/cases.cpp backend/agent.cpp backend/evidence.cpp backend/http.cpp backend/models.cpp)
target_include_directories(romeo PRIVATE core)
target_include_directories(romeo SYSTEM PRIVATE third_party/stb third_party/quill/include)
target_compile_definitions(romeo PRIVATE QUILL_NO_EXCEPTIONS)
target_compile_options(romeo PRIVATE -O2)
target_link_libraries(romeo PRIVATE yyjson CURL::libcurl Threads::Threads)
